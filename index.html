document.addEventListener('DOMContentLoaded', () => { // Referencias de elementos de la interfaz original const homeView = document.getElementById('homeView'); const chatView = document.getElementById('chatView'); const chatMessages = document.getElementById('chatMessages'); const homeChatInput = document.getElementById('homeChatInput'); const homeSendBtn = document.getElementById('homeSendBtn'); const chatChatInput = document.getElementById('chatChatInput'); const chatSendBtn = document.getElementById('chatSendBtn'); let currentHistory = []; async function handleSend(inputElement) { if (!inputElement) return; const text = inputElement.value.trim(); if (!text) return; // Limpiar input inputElement.value = ''; // Mostrar vista de chat si estamos en el inicio if (homeView && chatView) { homeView.style.display = 'none'; chatView.style.display = 'flex'; } // Agregar mensaje del usuario a la pantalla appendMessage(text, 'user'); // Llamar a Ollama local await callOllama(text); } function appendMessage(text, sender) { if (!chatMessages) return; const messageDiv = document.createElement('divclassName'); // corregido abajo const div = document.createElement('div'); div.style.margin = '12px 0'; div.style.padding = '10px 14px'; div.style.borderRadius = '8px'; div.style.maxWidth = '80%'; div.style.wordBreak = 'break-word'; if (sender === 'user') { div.style.background = '#2b2b2b'; div.style.marginLeft = 'auto'; div.textContent = text; } else { div.style.background = '#1e1e1e'; div.style.border = '1px solid #333'; div.style.marginRight = 'auto'; div.textContent = text; } chatMessages.appendChild(div); chatMessages.scrollTop = chatMessages.scrollHeight; } async function callOllama(text) { // Indicador de "Pensando..." const loadingDiv = document.createElement('div'); loadingDiv.id = 'loadingIndicator'; loadingDiv.style.margin = '12px 0'; loadingDiv.style.color = '#888'; loadingDiv.textContent = 'Llama 3 está pensando...'; chatMessages.appendChild(loadingDiv); chatMessages.scrollTop = chatMessages.scrollHeight; try { const response = await fetch('http://localhost:11434/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ model: 'llama3', messages: [{ role: 'user', content: text }], stream: false }) }); const data = await response.json(); document.getElementById('loadingIndicator')?.remove(); if (data.message && data.message.content) { appendMessage(data.message.content, 'ai'); } else { appendMessage('No se obtuvo respuesta de Llama 3.', 'ai'); } } catch (error) { document.getElementById('loadingIndicator')?.remove(); appendMessage(`Error de conexión con Ollama local: Asegúrate de tener la terminal abierta. (${error.message})`, 'ai'); } } // Eventos para el botón y la tecla Enter en la pantalla de inicio if (homeSendBtn && homeChatInput) { homeSendBtn.addEventListener('click', () => handleSend(homeChatInput)); homeChatInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') handleSend(homeChatInput); }); } // Eventos para el botón y la tecla Enter en la pantalla de chat activo if (chatSendBtn && chatChatInput) { chatSendBtn.addEventListener('click', () => handleSend(chatChatInput)); chatChatInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') handleSend(chatChatInput); }); } });
